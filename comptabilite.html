LIVRE DES RECETTES ET DÉPENSES (obligatoire, chronologique) -- --------------------------------------------------- CREATE TABLE IF NOT EXISTS livre_comptable ( id INT AUTO_INCREMENT PRIMARY KEY, account_id INT NOT NULL, date_operation DATE NOT NULL, type ENUM('recette','depense') NOT NULL, categorie VARCHAR(100) NOT NULL, sous_categorie VARCHAR(100) DEFAULT NULL, libelle VARCHAR(255) NOT NULL, montant_ht DECIMAL(10,2) NOT NULL, justificatif_url VARCHAR(500) DEFAULT NULL, mode_paiement ENUM('stripe','klarna','virement','especes','autre') DEFAULT 'virement', reference_piece VARCHAR(100) DEFAULT NULL, notes TEXT DEFAULT NULL, created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP, updated_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP ON UPDATE CURRENT_TIMESTAMP, INDEX idx_account_date (account_id, date_operation), INDEX idx_type (type), INDEX idx_categorie (categorie) ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4; -- Catégories RECETTES autorisées (référence, non contraignant en DB): -- Consultations Roqya, Diagnostics, Formations/Academy, -- Produits/supports numériques, Autres prestations spirituelles -- Catégories DÉPENSES autorisées (référence, non contraignant en DB): -- Matériel/équipement, Logiciels/SaaS, Marketing/publicité, -- Déplacement/véhicule, Formation professionnelle, -- Frais bancaires/commissions, Loyer/local, Assurances professionnelles, -- AVS/cotisations sociales, Téléphonie/internet, Fournitures de bureau, -- Frais de représentation/networking, Cotisations professionnelles, -- Autres frais généraux -- --------------------------------------------------- -- 2. AMORTISSEMENTS (matériel/équipement — suivi séparé) -- --------------------------------------------------- CREATE TABLE IF NOT EXISTS amortissements ( id INT AUTO_INCREMENT PRIMARY KEY, account_id INT NOT NULL, designation VARCHAR(255) NOT NULL, date_achat DATE NOT NULL, valeur_achat DECIMAL(10,2) NOT NULL, taux_amortissement DECIMAL(5,2) NOT NULL DEFAULT 30.00, duree_vie_estimee_annees INT DEFAULT 3, livre_comptable_id INT DEFAULT NULL COMMENT 'lien vers dépense origine', created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP, FOREIGN KEY (livre_comptable_id) REFERENCES livre_comptable(id) ON DELETE SET NULL, INDEX idx_account (account_id) ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4; -- --------------------------------------------------- -- 3. ÉTAT DU PATRIMOINE (snapshot annuel obligatoire au 31/12) -- --------------------------------------------------- CREATE TABLE IF NOT EXISTS patrimoine_annuel ( id INT AUTO_INCREMENT PRIMARY KEY, account_id INT NOT NULL, annee_fiscale INT NOT NULL, date_etablissement DATE NOT NULL, -- ACTIFS solde_bancaire_pro DECIMAL(10,2) NOT NULL DEFAULT 0, valeur_materiel_residuelle DECIMAL(10,2) NOT NULL DEFAULT 0, creances_clients DECIMAL(10,2) NOT NULL DEFAULT 0, -- PASSIFS dettes_fournisseurs DECIMAL(10,2) NOT NULL DEFAULT 0, avances_clients_non_facturees DECIMAL(10,2) NOT NULL DEFAULT 0, prets DECIMAL(10,2) NOT NULL DEFAULT 0, notes TEXT DEFAULT NULL, created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP, UNIQUE KEY unique_account_annee (account_id, annee_fiscale) ) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4;
